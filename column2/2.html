Visibility of Variables: var, let, and const

			var globalVar = "I am global!";

			console.log(globalVar); // Accessible: "I am global!"
			console.log(window.globalVar); // Accessible: "I am global!"
			
			var greeter = "hey hi";
			var greeter = "say Hello instead"; // Redeclared

			console.log(greeter); // "say Hello instead"

			and this also won't get an error

			var greeter = "hey hi";
			greeter = "say Hello instead"; // Redeclared

			console.log(greeter); // "say Hello instead"
			
			let greeting = "say Hi";
			greeting = "say Hello instead";

			this will return an error:

			let greeting = "say Hi";
			let greeting = "say Hello instead"; // error: Identifier 'greeting' has already been declared
			
			const greeting = "say Hi";
			greeting = "say Hello instead"; // error: Assignment to constant variable.

			nor this:

			const greeting = "say Hi";
			const greeting = "say Hello instead"; // error: Identifier 'greeting' has already been declared
			
			function testVar() {
				if (true) {
					var x = 10; // Function-scoped
				}
				console.log(x); // Accessible here: 10
			}
			testVar();
			
			function testLet() {
				if (true) {
					let x = 20; // Block-scoped
					console.log(x); // Accessible here: 20
				}
				// console.log(x); // ReferenceError: x is not defined
			}
			testLet();
			
			function testConst() {
				if (true) {
					const x = 30; // Block-scoped
					console.log(x); // Accessible here: 30
				}
				// console.log(x); // ReferenceError: x is not defined
			}
			testConst();
			
			class Example {
				method() {
					if (true) {
						var a = "var in method"; // Function-scoped
						let b = "let in method"; // Block-scoped
						const c = "const in method"; // Block-scoped
					}
					console.log(a); // Accessible: "var in method"
					// console.log(b); // ReferenceError
					// console.log(c); // ReferenceError
				}
			}
			
			const obj = new Example();
			obj.method();
			
			class Example {
				// Public property
				publicProperty = "This is public";
				
				// Private property
				#privateProperty = "This is private";
			
				getPrivateProperty() {
					return this.#privateProperty;
				}
			}
			
			const obj = new Example();
			console.log(obj.publicProperty); // Accessible: "This is public"
			// console.log(obj.#privateProperty); // SyntaxError: Private field must be accessed in class
			console.log(obj.getPrivateProperty()); // Accessible via getter: "This is private"
			
			class Example {
				static staticProperty = "This is static";
			
				static getStaticProperty() {
					return Example.staticProperty;
				}
			}
			
			console.log(Example.staticProperty); // Accessible directly: "This is static"
			console.log(Example.getStaticProperty()); // Accessible via static method: "This is static"
